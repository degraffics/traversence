278:     /**
279:      * One batch of one kind. $after is the last id done ('' to start). Returns ['done' => rows looked at, 'next' => id or
280:      * null when the kind is finished]. A full build removes, at its end, rows that weren't seen (deleted items).
281:      */
282:     public static function build(PDO $pdo, string $kind, string $after = '', int $limit = 400): array
283:     {
284:         if ($after === '') self::setState($pdo, 'build_' . $kind . '_started', date('Y-m-d H:i:s'));
285:         $limit = max(50, min(1000, $limit));
286:         switch ($kind) {
287:             case 'listing':  [$n, $last] = self::listingsAfter($pdo, (int) $after, $limit); break;
288:             case 'outdoors': [$n, $last] = self::outdoorsAfter($pdo, $after, $limit); break;
289:             case 'story': case 'group': case 'topic': [$n, $last] = [self::rebuildSmall($pdo, $kind), null]; break;

297:             if ($kind === 'listing') self::setState($pdo, 'listing_mark', $since !== '' ? $since : date('Y-m-d H:i:s'));

569:     public static function refresh(PDO $pdo, bool $force = false): void
570:     {
571:         try {
572:             if (!self::ready($pdo)) return;
573:             $last = self::state($pdo, 'refreshed_at');
574:             if (!$force && $last !== '' && strtotime($last) > time() - 900) return;
575:             self::setState($pdo, 'refreshed_at', date('Y-m-d H:i:s'));
576:             $mark = self::state($pdo, 'listing_mark');
577:             if ($mark !== '') {
578:                 // listings changed since the mark, oldest first, 600 at a time. The mark moves to the last one taken, so each
579:                 // run carries on where the last stopped (2026-11-10: it moved only when fewer than 300 had changed, so with
580:                 // the crawler busy it re-read the same 300 every time and new listings, Dana Point's, never reached search)
581:                 $now = date('Y-m-d H:i:s'); $n = 600;
582:                 $st = $pdo->prepare('SELECT id, updated_at FROM entities WHERE updated_at > ? ORDER BY updated_at, id LIMIT ' . $n);
583:                 $st->execute([$mark]); $er = $st->fetchAll(PDO::FETCH_ASSOC);
584:                 $st = $pdo->prepare('SELECT entity_id, MAX(updated_at) u FROM entity_metadata WHERE updated_at > ? GROUP BY entity_id ORDER BY u LIMIT ' . $n);
585:                 $st->execute([$mark]); $mr = $st->fetchAll(PDO::FETCH_ASSOC);
586:                 $ids = array_values(array_unique(array_merge(array_map(fn ($r) => (int) $r['id'], $er), array_map(fn ($r) => (int) $r['entity_id'], $mr))));
587:                 if ($ids) foreach (array_chunk($ids, 300) as $chunk) self::indexListings($pdo, $chunk);
588:                 // as far as both lists reached: a full list stops at its last row; one with room left reached now
589:                 $back = fn (string $t) => date('Y-m-d H:i:s', strtotime($t) - 1);   // a second back: others changed that same second are read again
590:                 $reach = [count($er) >= $n ? $back((string) end($er)['updated_at']) : $now, count($mr) >= $n ? $back((string) end($mr)['u']) : $now];
591:                 $next = min($reach);
592:                 if ($next <= $mark) $next = date('Y-m-d H:i:s', strtotime($mark) + 1);   // a second full of changes: step past it
593:                 self::setState($pdo, 'listing_mark', $next);
594:             }
595:             foreach (['story', 'group', 'topic'] as $k) self::rebuildSmall($pdo, $k);
596:             // landmarks added after the federal data (by hand, or found for a guide on demand: ids from 900000000), new since
597:             // last time; they were only indexed on a full rebuild (2026-11-10: Four Corners Monument, Dana Point's places)
598:             if (class_exists('Landmarks') || is_file(__DIR__ . '/Landmarks.php')) {
599:                 require_once __DIR__ . '/Landmarks.php';
600:                 if (Landmarks::ready($pdo)) {
601:                     $lm = self::state($pdo, 'landmark_added_mark');
602:                     if ($lm === '') $lm = 'g' . str_pad((string) (Landmarks::MANUAL_FROM - 1), 10, '0', STR_PAD_LEFT);
603:                     [$n, $next] = self::landmarksAfter($pdo, $lm, 300);
604:                     if ($n > 0) self::setState($pdo, 'landmark_added_mark', $next);
605:                     // each landmark filed under one of our categories by its source's class (2026-11-10: "from categories not
606:                     // guesses"); a few filed now are re-indexed at once, many go through the walk below
607:                     $filed = Landmarks::categorize($pdo);
608:                     if (count($filed) > 300) { self::setState($pdo, 'landmark_kind_walk3', 'g'); }
609:                     else foreach ($filed as $fid) self::refreshLandmark($pdo, (int) $fid);
610:                     // every landmark again under its kind, 1000 a run until done (once, and after a big filing)
611:                     $wk = self::state($pdo, 'landmark_kind_walk3');   // 3: with its activities (Landmarks::activities, 2026-11-10)
612:                     if ($wk === '') $wk = 'g';
613:                     if ($wk !== 'done') { [$wn, $wnext] = self::landmarksAfter($pdo, $wk, 1000); self::setState($pdo, 'landmark_kind_walk3', $wn < 1000 ? 'done' : $wnext); }
614:                 }
615:             }
616:         } catch (Throwable $e) {
617:             error_log('SearchIndex::refresh: ' . $e->getMessage());
618:         }
619:     }

800:         $total = count($out);

807:             $per[$x['kind']] = ($per[$x['kind']] ?? 0) + 1;
808:             unset($x['score']);
809:             $items[] = $x;
810:         }
811:         return ['items' => $items, 'kinds' => $kinds, 'funnels' => $funnels, 'audience_found' => $audFound, 'total' => $total, 'loose' => $any];

866:     public static function miles(float $a, float $b, float $c, float $d): float
867:     {
868:         $r = M_PI / 180;
869:         $x = sin(($c - $a) * $r / 2) ** 2 + cos($a * $r) * cos($c * $r) * sin(($d - $b) * $r / 2) ** 2;
870:         return 3958.8 * 2 * asin(min(1, sqrt($x)));
871:     }

873:     /** SQL on si.* for a place (boundary) or an area (a box; the exact distance is checked after). '' = anywhere. */
874:     public static function whereSql(PDO $pdo, ?array $place, ?array $area, array &$p): string
875:     {
876:         if ($area) {
877:             $dLat = $area['radius'] / 69.0; $dLon = $area['radius'] / max(0.0001, 69.0 * cos(deg2rad((float) $area['lat'])));
878:             $p += [':la1' => $area['lat'] - $dLat, ':la2' => $area['lat'] + $dLat, ':lo1' => $area['lon'] - $dLon, ':lo2' => $area['lon'] + $dLon];
879:             return '(si.lat BETWEEN :la1 AND :la2 AND si.lon BETWEEN :lo1 AND :lo2)';
880:         }
881:         if (!$place) return '';
882:         $or = [];
883:         $param = ['cluster' => 'cluster', 'geo_hub' => 'geo', 'hub' => 'hub'][$place['level'] ?? ''] ?? null;
884:         if ($param && !empty($place['slug']) && empty($place['town'])) {   // a town in its area: its own ZIPs only
885:             $s = PlaceScope::resolve($pdo, [$param => (string) $place['slug']]);
886:             if ($s && $s['found']) $or[] = str_replace(['e.micro_cluster_id', 'e.primary_zip'], ['si.micro_cluster_id', 'si.zip'], (string) $s['sql']);
887:         }
888:         $zips = array_slice(array_values(array_filter(array_map('strval', (array) ($place['zips'] ?? [])))), 0, 60);
889:         // a county: every ZIP in it, not only the first 60 ("Los Angeles County" has hundreds)
890:         if (!empty($place['county']) && !empty($place['state'])) {
891:             $or[] = 'si.zip IN (SELECT TRIM(zc.zip) FROM zip_coordinates zc WHERE TRIM(zc.county) = :pco AND TRIM(zc.state) = :pcs)';
892:             $p[':pco'] = (string) $place['county']; $p[':pcs'] = (string) $place['state'];
893:             $zips = [];
894:         }
895:         // a state: every ZIP in it
896:         if (!empty($place['is_state']) && !empty($place['state'])) {
897:             $or[] = 'si.zip IN (SELECT TRIM(zc.zip) FROM zip_coordinates zc WHERE TRIM(zc.state) = :pst)';
898:             $p[':pst'] = (string) $place['state'];
899:             $zips = [];
900:         }
901:         // a town: every ZIP in it, and listings that give it as their city, not only its first 60 ZIPs ("pizza in phoenix":
902:         // Phoenix has more than 60, and a downtown pizzeria's ZIP was past them)
903:         if (!empty($place['city']) && !empty($place['state']) && empty($place['county']) && empty($place['is_state'])) {
904:             $or[] = 'si.zip IN (SELECT TRIM(zc.zip) FROM zip_coordinates zc WHERE zc.city = :ptc AND TRIM(zc.state) = :pts)';
905:             $or[] = '(LOWER(si.city) = :ptc2 AND si.state = :pts2)';
906:             $p[':ptc'] = (string) $place['city']; $p[':pts'] = (string) $place['state'];
907:             $p[':ptc2'] = mb_strtolower(trim((string) $place['city'])); $p[':pts2'] = (string) $place['state'];
908:         }
909:         if ($zips) {
910:             $ph = [];
911:             foreach ($zips as $i => $z) { $ph[] = ":z$i"; $p[":z$i"] = substr(trim($z), 0, 5); }
912:             $or[] = 'si.zip IN (' . implode(',', $ph) . ')';
913:         }
914:         if (!empty($place['key'])) { $or[] = 'si.place_key = :pkey'; $p[':pkey'] = (string) $place['key']; }
915:         if (!$or && !empty($place['city'])) { $or[] = '(LOWER(si.city) = :pcity AND si.state = :pst)'; $p[':pcity'] = mb_strtolower((string) $place['city']); $p[':pst'] = (string) ($place['state'] ?? ''); }
916:         if (!$or && !empty($place['state'])) { $or[] = 'si.state = :pst2'; $p[':pst2'] = (string) $place['state']; }
917:         return $or ? '(' . implode(' OR ', $or) . ')' : '';
918:     }
