110:     /**
111:      * Why the team's own listing isn't on the public site, in words for them; null when it's public.
112:      * Said plainly only where the reason is theirs to know (the test listing, a private client record, merged into
113:      * another listing, not public yet). Anything held by staff (a data check, low trust, Restricted, a nation's or a
114:      * sensitive place's hold: decisions/0066, 0164 R7) is only "under review", with Ask our team: never the staff reason.
115:      * A merged listing also carries 'merged' => [kept id, kept name]: changes belong on the kept one, so the editor locks.
116:      * @return array{text: string, link: ?array{0: string, 1: string}, ask: bool, merged?: array{0: int, 1: string}}|null
117:      */
118:     public static function hiddenReason(PDO $pdo, array $row): ?array
119:     {
120:         $e = (int) ($row['id'] ?? 0);
121:         $public = ['unclaimed', 'claimed-unverified', 'pending-verification', 'verified'];
122:         if ((int) ($row['is_suppressed'] ?? 0) === 0 && in_array((string) ($row['status'] ?? ''), $public, true)) return null;
123:         $review = ['text' => 'It\'s under review by our team, so it isn\'t on the public site or in search for now. You can keep editing it; your changes show once it\'s back.', 'link' => null, 'ask' => true];
124:         if ((int) ($row['is_suppressed'] ?? 0) === 0)
125:             return ['text' => 'It isn\'t public yet: our team checks a new or changed listing before it shows on the site and in search. You can keep editing it meanwhile.', 'link' => null, 'ask' => true];
126:         try {   // the hidden test listing for the site's checks (TestAccounts)
127:             $st = $pdo->prepare("SELECT 1 FROM test_records WHERE kind = 'listing' AND ref_id = ? LIMIT 1"); $st->execute([$e]);
128:             if ($st->fetchColumn()) return ['text' => 'This is the test listing for the site\'s checks. It\'s never shown on the public site or in search.', 'link' => null, 'ask' => false];
129:         } catch (Throwable) {}
130:         try {   // a private client record (Clients::makePrivate): kept for orders, never public
131:             $st = $pdo->prepare('SELECT 1 FROM client_listings WHERE entity_id = ? AND private = 1 LIMIT 1'); $st->execute([$e]);
132:             if ($st->fetchColumn()) return ['text' => 'It\'s kept private: our team holds it for your orders with us, and it isn\'t shown on the public site or in search. To list your business publicly, ask our team.', 'link' => null, 'ask' => true];
133:         } catch (Throwable) {}
134:         try {   // merged into another listing of the same business (decisions/0072): its link takes visitors there
135:             $st = $pdo->prepare("SELECT meta_value FROM entity_metadata WHERE entity_id = ? AND meta_key = 'same_as'"); $st->execute([$e]);
136:             $same = json_decode((string) $st->fetchColumn(), true);

142:                                              'link' => ['/listing/view.php?id=' . $keep, 'See ' . (string) $kept], 'ask' => true, 'merged' => [$keep, (string) $kept]];
143:             }
144:         } catch (Throwable) {}
145:         return $review;

247:      * Confirms the email code: the listing becomes Managed by owner, the claim gets its text code and a 48-hour due

251:     {

255:         catch (RuntimeException $x) { throw new InvalidArgumentException($x->getMessage()); }
256:         $l = self::listing($pdo, $e);
257:         if (!$l) throw new InvalidArgumentException('Listing not found.');
258:         $id = 0;

784:     public static function fileDispute(PDO $pdo, array $user, int $e, string $reason): int
785:     {

791:         if (mb_strlen(trim($reason)) < 10) throw new InvalidArgumentException('Tell us why, in a sentence or two.');

836:     /** Admin: accept (freeze edits while looking), keep the owner, transfer to the challenger, or return to unclaimed. */
837:     public static function decideDispute(PDO $pdo, int $disputeId, int $adminId, string $what, string $note = ''): void
838:     {
839:         $st = $pdo->prepare("SELECT * FROM disputes WHERE id = ? AND status = 'open'");
840:         $st->execute([$disputeId]);
841:         $d = $st->fetch(PDO::FETCH_ASSOC);
842:         if (!$d) throw new InvalidArgumentException('This dispute is already closed.');
843:         $e = (int) $d['entity_id'];
844:         if ($what === 'accept') {
845:             $pdo->prepare('UPDATE disputes SET accepted_at = NOW(), admin_note = CONCAT_WS(\'\n\', admin_note, ?) WHERE id = ?')->execute([$note ?: null, $disputeId]);
846:             return;
847:         }
848:         if (!in_array($what, ['kept', 'transferred', 'returned'], true)) throw new InvalidArgumentException('Pick a decision.');
849:         $h = self::holding($pdo, $e);
850:         $ch = $d['claim_id'] ? self::get($pdo, (int) $d['claim_id']) : null;
851:         Database::transaction(function (PDO $pdo) use ($what, $h, $ch, $e, $adminId, $note, $disputeId) {
852:             if ($what !== 'kept' && $h) self::release($pdo, $h, 'declined', $adminId, 'Dispute #' . $disputeId . ': ' . $what);
853:             if ($what === 'transferred' && $ch) {
854:                 $l = self::listing($pdo, $e);
855:                 $pdo->prepare("UPDATE listing_claims SET status = 'withdrawn', decided_by = ?, decided_at = NOW() WHERE id = ?")->execute([$adminId, $ch['id']]);
856:                 $id = self::open($pdo, $e, (int) $ch['user_id'], null, (string) $ch['role_note'], (string) $ch['claimant_phone'], (string) ($l['phone'] ?? ''));
857:                 $pdo->prepare("UPDATE listing_claims SET doc_path = ?, doc_name = ? WHERE id = ?")->execute([$ch['doc_path'], $ch['doc_name'], $id]);
858:             } elseif ($ch && $ch['status'] === 'on_hold') {
859:                 $pdo->prepare("UPDATE listing_claims SET status = 'declined', decided_by = ?, decided_at = NOW() WHERE id = ?")->execute([$adminId, $ch['id']]);
860:             }
861:             $pdo->prepare('UPDATE disputes SET status = ?, decided_by = ?, resolved_at = NOW(), admin_note = CONCAT_WS(\'\n\', admin_note, ?) WHERE id = ?')
862:                 ->execute([$what, $adminId, $note ?: null, $disputeId]);
863:         });
864:         if ($what === 'transferred' && $ch) {
865:             $new = self::holding($pdo, $e);
866:             if ($new) self::verify($pdo, (int) $new['id'], $adminId, 'admin', 'Transferred after dispute #' . $disputeId . '.');
867:         }
868:         self::purgeDocuments($pdo);
869:         $msg = ['kept' => 'The listing stays with its current owner.', 'transferred' => 'The listing was transferred to the person who filed the dispute.', 'returned' => 'The listing returned to unclaimed; anyone can claim it again.'][$what];
870:         if ($h) self::tellClaimant($pdo, $h, 'Dispute decided', $msg . ($note !== '' ? "\n\n" . $note : ''));
871:         if ($ch) self::tellClaimant($pdo, $ch, 'Dispute decided', $msg . ($note !== '' ? "\n\n" . $note : ''));
872:     }
