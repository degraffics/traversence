72:     /** The visitor's choices for the cookie tools, from the tv_choices cookie: tool => true|false (absent: not asked). */
73:     public const COOKIE = 'tv_choices';
74: 
75:     public static function cookieChoices(): array
76:     {
77:         $raw = json_decode((string) ($_COOKIE[self::COOKIE] ?? ''), true);
78:         $out = [];
79:         foreach ((array) $raw as $tool => $v) {
80:             if (!isset(self::TOOLS[$tool]) || empty(self::TOOLS[$tool]['cookie']) || !is_array($v)) continue;
81:             if ((int) ($v[1] ?? 0) >= self::TOOLS[$tool]['version']) $out[$tool] = (bool) ($v[0] ?? false);   // an older wording asks again
82:         }
83:         return $out;
84:     }
85: 
86:     /** Remembers a visitor's choice in tv_choices (first-party, a year; only the choices, readable by the page's script). */
87:     public static function setCookieChoice(string $tool, bool $on): void
88:     {
89:         if (!isset(self::TOOLS[$tool]) || empty(self::TOOLS[$tool]['cookie'])) throw new InvalidArgumentException('Unknown choice.');
90:         $all = (array) json_decode((string) ($_COOKIE[self::COOKIE] ?? ''), true);
91:         $all[$tool] = [$on ? 1 : 0, self::TOOLS[$tool]['version'], date('Ymd')];
92:         $val = json_encode($all);
93:         $_COOKIE[self::COOKIE] = $val;
94:         if (!headers_sent()) setcookie(self::COOKIE, $val, ['expires' => time() + 365 * 86400, 'path' => '/', 'secure' => !empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off', 'httponly' => false, 'samesite' => 'Lax']);
95:     }

97:     /**
98:      * Whether a cookie tool is allowed for this visit: a signed-in person's account choice wins, then this browser's.
99:      * Never asked means no.
100:      */
101:     public static function allows(?PDO $pdo, ?int $uid, string $tool): bool
102:     {
103:         if ($uid && $pdo && self::ready($pdo)) {
104:             $r = self::row($pdo, $uid, $tool);
105:             if ($r !== null) return $r['status'] === 'active' && (int) $r['version'] >= self::TOOLS[$tool]['version'];
106:         }
107:         return self::cookieChoices()[$tool] ?? false;
108:     }
109: 
110:     public static function ready(PDO $pdo): bool
111:     {
112:         try { $pdo->query('SELECT 1 FROM tool_consent_log LIMIT 1'); $pdo->query('SELECT 1 FROM tool_consents LIMIT 1'); return true; } catch (PDOException) { return false; }
113:     }
114: 
115:     private static function row(PDO $pdo, int $uid, string $tool): ?array
116:     {
117:         $st = $pdo->prepare('SELECT status, version, activated_at, deactivated_at FROM tool_consents WHERE user_id = ? AND tool = ?');
118:         $st->execute([$uid, $tool]);
119:         return $st->fetch(PDO::FETCH_ASSOC) ?: null;
120:     }
121: 
122:     /** Active with the current wording? (A tool with an older version needs consent again.) */
123:     public static function active(PDO $pdo, int $uid, string $tool): bool
124:     {
125:         if (!isset(self::TOOLS[$tool]) || !self::ready($pdo)) return false;
126:         $r = self::row($pdo, $uid, $tool);
127:         return $r !== null && $r['status'] === 'active' && (int) $r['version'] >= self::TOOLS[$tool]['version'];
128:     }
129: 
130:     /** Switched off on purpose (as opposed to never asked): the tool's effects are paused. */
131:     public static function paused(PDO $pdo, int $uid, string $tool): bool
132:     {
133:         if (!self::ready($pdo)) return false;
134:         $r = self::row($pdo, $uid, $tool);
135:         return $r !== null && $r['status'] === 'inactive';
136:     }

138:     public static function set(PDO $pdo, int $uid, string $tool, bool $on, string $page = '', bool $fullShown = false): void
139:     {
140:         if (!isset(self::TOOLS[$tool])) throw new InvalidArgumentException('Unknown tool.');
141:         $v = self::TOOLS[$tool]['version'];
142:         if ($on && self::TOOLS[$tool]['read_required'] && !$fullShown) throw new InvalidArgumentException('Please read the full text first.');
143:         $now = date('Y-m-d H:i:s');
144:         if (self::row($pdo, $uid, $tool) === null) {
145:             $pdo->prepare('INSERT INTO tool_consents (user_id, tool, version, status, activated_at, deactivated_at, updated_at) VALUES (?, ?, ?, ?, ?, ?, ?)')
146:                 ->execute([$uid, $tool, $v, $on ? 'active' : 'inactive', $on ? $now : null, $on ? null : $now, $now]);
147:         } elseif ($on) {
148:             $pdo->prepare("UPDATE tool_consents SET status = 'active', version = ?, activated_at = ?, updated_at = ? WHERE user_id = ? AND tool = ?")->execute([$v, $now, $now, $uid, $tool]);
149:         } else {
150:             $pdo->prepare("UPDATE tool_consents SET status = 'inactive', deactivated_at = ?, updated_at = ? WHERE user_id = ? AND tool = ?")->execute([$now, $now, $uid, $tool]);
151:         }
152:         $pdo->prepare('INSERT INTO tool_consent_log (user_id, tool, version, action, full_text_shown, page, created_at) VALUES (?, ?, ?, ?, ?, ?, ?)')
153:             ->execute([$uid, $tool, $v, $on ? 'activate' : 'deactivate', $fullShown ? 1 : 0, mb_substr($page, 0, 255), $now]);
154:         self::keepWording($pdo, $tool);
155:         if (!empty(self::TOOLS[$tool]['cookie'])) self::setCookieChoice($tool, $on);   // this browser follows the account
156:     }

158:     /** The exact wording of each version, kept the first time it's agreed to (decisions/0062 §4), so the log's version
159:      *  number always leads back to the words the person saw. */
160:     private static function keepWording(PDO $pdo, string $tool): void
161:     {
162:         $t = self::TOOLS[$tool];
163:         try {
164:             $st = $pdo->prepare('SELECT 1 FROM consent_wordings WHERE tool = ? AND version = ?'); $st->execute([$tool, $t['version']]);
165:             if ($st->fetchColumn()) return;
166:             $pdo->prepare('INSERT INTO consent_wordings (tool, version, summary, explanation, first_used) VALUES (?, ?, ?, ?, ?)')
167:                 ->execute([$tool, $t['version'], $t['summary'], $t['explanation'], date('Y-m-d H:i:s')]);
168:         } catch (PDOException) { /* not set up yet (2026-11-01_verification_consent.sql), or two at once */ }
169:     }

171:     /** Your tools: each tool's wording and this person's state. */
172:     public static function all(PDO $pdo, int $uid): array
173:     {
174:         $out = [];
175:         foreach (self::TOOLS as $key => $t) {
176:             $r = self::row($pdo, $uid, $key);
177:             $state = $r === null ? 'not_asked' : ($r['status'] === 'active' ? ((int) $r['version'] >= $t['version'] ? 'active' : 'update') : 'paused');
178:             $out[] = ['tool' => $key, 'name' => $t['name'], 'summary' => $t['summary'], 'explanation' => $t['explanation'], 'paused_note' => $t['paused'],
179:                       'declined_note' => $t['declined'] ?? '', 'read_required' => $t['read_required'], 'state' => $state,
180:                       'since' => $r ? ($r['status'] === 'active' ? $r['activated_at'] : $r['deactivated_at']) : null];
181:         }
182:         return $out;
183:     }
184: 
185:     /** What the consent chip needs for one tool. */
186:     public static function prompt(string $tool): array
187:     {
188:         $t = self::TOOLS[$tool];
189:         return ['tool' => $tool, 'name' => $t['name'], 'summary' => $t['summary'], 'explanation' => $t['explanation'], 'read_required' => $t['read_required']];
190:     }
