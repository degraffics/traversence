468:     private static function seed(PDO $pdo): void
469:     {
470:         $have = class_exists('SearchIndex') && SearchIndex::ready($pdo) ? SearchIndex::state($pdo, 'situations_seed') : '';
471:         $count = (int) $pdo->query('SELECT COUNT(*) FROM situations')->fetchColumn();
472:         if ($count > 0 && $have === self::SEED_VERSION) return;
473:         $s = $pdo->prepare('INSERT INTO situations (slug, label, urgency, danger, track, status, source) VALUES (?, ?, ?, ?, ?, ?, ?)');
474:         $t = $pdo->prepare('INSERT INTO situation_triggers (situation_id, phrase) VALUES (?, ?)');

477:         $pdo->beginTransaction();
478:         try {
479:             foreach (self::SEED as [$slug, $label, $urg, $danger, $track, $triggers, $needs]) {
480:                 $find->execute([$slug]);
481:                 $old = $find->fetch(PDO::FETCH_ASSOC);
482:                 if ($old && $old['source'] !== 'seed') continue;   // edited in Admin: theirs now
483:                 if ($old) {
484:                     $id = (int) $old['id'];

755:     /** Positions of $ph's words in $tok, in order, at most two words apart, not already used; or null. */

757:     {

765:                 if ($hit === null) { $ok = false; break; }
