2: // api/lib/SituationReasoner.php — reading a situation from what the words mean, not from a list of phrases
3: // (decisions/0061, the meaning layer). Rule-based, no AI.
4: //
5: // People describe the same thing a hundred ways ("bit by something", "a dog bit me", "got stung", "my ankle is
6: // swollen"). Instead of listing every wording, each word has a meaning class (bite, bit, bitten, stung → harm; dog,
7: // snake, "something" → creature; ankle → body; pipe → plumbing), a few grammar cues are read ("by" names what did it,
8: // "my" marks what's theirs, "out of" marks what's missing), and event frames turn the meaning into a situation:
9: // harm + creature → An injury; broken + plumbing → Leak or burst pipe; out of + gas → Gas for the car.
10: //
11: // Five kinds of word (the Traversence intent taxonomy, decisions/0061):
12: //   event   what happened (broke, stuck, bit)                         → Why: a situation, urgent ones first
13: //   thing   what it happened to, or what did it (wallet, pipe, dog)   → What / Who: what's looked up
14: //   action  how something is done (fix, install, register, cook)     → Why: who does it (how-to guides are coming)
15: //   tool    the tool or medium (generator, saw, app, permit portal)  → What: narrows what to find
16: //   time    when (asap, upcoming, this morning, fall)                 → When: open now, coming up, a season
17: //   modifier what kind of it (heavy-duty, local, used)                → narrows What: those that mention it first
18: //   wrapper  the shape of the ask (how to, where can I)              → structure: "how to" marks a procedure
19: // Every word also has a lexical category (noun, verb…), decisions/0063; see api/lib/Lexicon.php.
20: //
21: // It runs only when no phrase in the situation map matched. Each answer carries its reasoning, which the search term
22: // analysis (Admin → Missed searches, ✦) shows, so a wrong reading can be seen and corrected.
23: declare(strict_types=1);

379:     /**
380:      * Each word's kind, for ✦ Analyze: [['word', 'kind', 'class', 'w']], in the order they're typed. A word can be
381:      * read as more than one kind ("oven": a thing, and a tool); each reading is listed.
382:      */
383:     public static function kinds(string $q): array
384:     {
385:         $text = ' ' . trim(preg_replace('/\s+/', ' ', mb_strtolower(preg_replace("/[^a-z0-9' ]+/i", ' ', $q)))) . ' ';
386:         $text = str_replace(["can't", "won't", "don't", "'"], ['cant', 'wont', 'dont', ''], $text);
387:         $out = [];
388:         $add = function (string $kind, array $byClass, bool $keyed) use ($text, &$out) {
389:             foreach ($byClass as $c => $forms) {
390:                 $list = $keyed ? array_keys($forms) : $forms;
391:                 usort($list, fn ($a, $b) => strlen((string) $b) <=> strlen((string) $a));
392:                 foreach ($list as $f) {
393:                     $at = strpos($text, ' ' . $f . ' ');
394:                     if ($at === false) continue;
395:                     $out[] = ['word' => (string) $f, 'kind' => $kind, 'class' => (string) $c, 'w' => self::KINDS[$kind][1], 'at' => $at];
396:                     break;
397:                 }
398:             }
399:         };
400:         $add('event', self::events(), true); $add('action', self::actions(), true);
401:         $add('thing', self::things(), false); $add('tool', self::tools(), false); $add('time', self::times(), false);
402:         $add('modifier', self::modifiers(), false); $add('wrapper', self::wrappers(), false);
403:         usort($out, fn ($a, $b) => $a['at'] <=> $b['at']);
404:         return array_map(fn ($x) => array_diff_key($x, ['at' => 1]), $out);
405:     }

439:     public static function infer(array $tok, bool $noDict = false): ?array
440:     {
441:         $text = ' ' . implode(' ', $tok) . ' ';
442:         // the events said, longest forms first ("broke down" before "broke"); a weak one needs a thing or a cue
443:         $events = [];
444:         $lead = [];
445:         foreach (self::events() as $class => $forms) $lead[] = ['event', $class, $forms];
446:         foreach (self::actions() as $class => $forms) $lead[] = ['action', $class, $forms];
447:         foreach ($lead as [$kind, $class, $forms]) {
448:             uksort($forms, fn ($a, $b) => strlen($b) <=> strlen($a));
449:             foreach ($forms as $f => $weak) {
450:                 if (!preg_match('/ ' . preg_quote(self::nf($f), '/') . ' /', $text, $m, PREG_OFFSET_CAPTURE)) continue;
451:                 foreach ($forms as $f2 => $w2) if (self::nf($f2) === self::nf($f) && strpos(' ' . implode(' ', $tok) . ' ', ' ' . $f2 . ' ') !== false) { $f = $f2; $weak = $w2; break; }   // as typed ("bite", not "bites")
452:                 $events[] = ['class' => $class, 'word' => $f, 'weak' => (bool) $weak, 'at' => $m[0][1], 'kind' => $kind];
453:                 break;
454:             }
455:         }
456:         // any form of a verb that always means the same ("robbing", "leaked"), by its base form (decisions/0064)
457:         if (is_file(__DIR__ . '/WordFamilies.php')) {
458:             require_once __DIR__ . '/WordFamilies.php';
459:             $have = array_column($events, 'class');
460:             foreach ($tok as $t) {
461:                 foreach (WordFamilies::bases($t) as $b) {
462:                     if (!isset(self::VERBS[$b])) continue;
463:                     [$k, $c] = self::VERBS[$b];
464:                     if (!in_array($c, $have, true)) { $events[] = ['class' => $c, 'word' => $t, 'weak' => false, 'at' => (int) strpos($text, ' ' . $t . ' '), 'kind' => $k]; $have[] = $c; }
465:                     break;
466:                 }
467:             }
468:         }
469:         if (!$events) return null;
470:         // "a bit", "a little bit": an amount, not a bite
471:         $events = array_values(array_filter($events, fn ($e) => !($e['word'] === 'bit' && preg_match('/ (a|little|quite a|tiny) bit /', $text))));
472:         // the things said, and what kind each is
473:         $things = [];
474:         foreach (array_merge(self::things(), self::tools()) as $class => $forms) {
475:             usort($forms, fn ($a, $b) => strlen($b) <=> strlen($a));
476:             // shown as the person wrote it: the form that is exactly a word typed ("dog", not "dogs"), else the dictionary's ("taxes", not "taxe")
477:             $hitF = null;
478:             foreach ($forms as $f) if (preg_match('/ ' . preg_quote(self::nf($f), '/') . ' /', $text)) { if ($hitF === null) $hitF = $f; if (in_array($f, $tok, true)) { $hitF = $f; break; } }
479:             if ($hitF !== null) $things[$class] = $hitF;
480:         }
481:         // a thing nobody listed, from the dictionary (decisions/0064): "a purse" is someone's belongings, "a hamster" a pet
482:         $dict = [];
483:         if (!$noDict && class_exists('WordFamilies')) {
484:             for ($i = 0, $n = count($tok); $i < $n; $i++) {
485:                 foreach ([3, 2, 1] as $len) {
486:                     if ($i + $len > $n) continue;
487:                     $ph = implode(' ', array_slice($tok, $i, $len));
488:                     $fam = WordFamilies::family($ph);
489:                     if ($fam === null || isset($things[$fam])) continue;
490:                     $things[$fam] = $ph; $dict[$fam] = true; break;
491:                 }
492:             }
493:         }
494:         $cue = (bool) array_intersect($tok, self::CUES);
495:         $mine = (bool) preg_match('/ (my|our) /', $text);
496:         $toMe = (bool) preg_match('/ (me|i|im|myself) /', $text);
497:         foreach (self::frames() as $fr) {
498:             [$ev, $need, $slug, $why] = $fr; $flag = $fr[4] ?? '';
499:             foreach ($events as $e) {
500:                 if ($e['class'] !== $ev) continue;
501:                 $hit = null;
502:                 foreach ($need as $nc) if (isset($things[$nc])) { $hit = $nc; break; }
503:                 if ($need && $hit === null) continue;
504:                 if (!$need && $e['weak'] && !$cue) continue;   // "cut" alone could be a haircut; "got cut" is a harm
505:                 if (!$need && $e['weak'] && $e['kind'] === 'action') continue;   // "make" or "set up" alone says nothing about what
506:                 // a pet that's sick or hurt is the pet's; "my dog bit me" is the person's injury
507:                 if ($flag === 'mine' && !$mine) continue;
508:                 if ($flag === 'not-mine' && $mine) continue;   // "found my keys": their own, found again; nothing to hand in
509:                 if ($flag === 'stranded' && !in_array($e['word'], ['wont start', 'died', 'broke down', 'dead', 'stopped'], true)) continue;   // "my car died" strands them; "car is making a noise" doesn't
510:                 if ($flag === 'mine-patient' && (!$mine || $toMe || preg_match('/ by (my|our) /', $text))) continue;   // "my dog got bit" is the dog's; "my dog bit me" is theirs
511:                 if ($hit === 'creature' && $things[$hit] === 'something' && $ev !== 'harm') continue;
512:                 $t = $hit !== null ? $things[$hit] : '';
513:                 $by = $ev === 'harm' && preg_match('/ by (a |an |the |some |my |our )?' . preg_quote(self::nf($t), '/') . ' /', $text) ? ' ("by" names what did it)' : '';
514:                 $fromDict = $hit !== null && isset($dict[$hit]);
515:                 return ['slug' => $slug, 'why' => str_replace(['{e}', '{t}'], [$e['word'], $t], $why) . $by . ($fromDict ? ' ("' . $t . '" is ' . (WordFamilies::LABEL[$hit] ?? $hit) . ', from the dictionary)' : ''),
516:                         'dict' => $fromDict, 'kind' => $e['kind'], 'thing' => $t,
517:                         'words' => array_values(array_filter(array_merge(explode(' ', self::nf($e['word'])), explode(' ', self::nf($t)), $by !== '' ? ['by'] : [])))];
518:             }
519:         }
520:         return null;
521:     }
