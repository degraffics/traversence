145:     // ---- Closing first: 30 days to change your mind (decisions/0091, 2026-10-15) ------------------------------------
146:     //
147:     // close()    what they posted is hidden (the way each kind is hidden, with what it was before kept in the closure's
148:     //            record), every session ends, and the account waits 30 days. Nothing is erased yet; a listing they run
149:     //            stays as it is. An email has a Restore link.
150:     //   restore()  puts back exactly what close() hid (only where nobody has changed it since), and the account is open
151:     //   sweep()    on day 30, run() does the full deletion. Page loads run it (tickSoon, at most every 30 minutes).
152:     // Signing in while it's closing (password, Google, Apple, Microsoft, Facebook) goes to /user/restore-account.php,
153:     // which offers Restore or Keep it closed; nothing else opens until then.

176:     /** Closes the account for 30 days. Returns ['email', 'delete_after', 'token']. Throws RuntimeException when blocked. */
177:     public static function close(PDO $pdo, int $uid): array
178:     {
179:         if ($b = self::blockers($pdo, $uid)) throw new RuntimeException($b[0][0]);
180:         if (!self::ready($pdo)) throw new RuntimeException('Closing accounts isn\'t set up yet (a database update). Try again later.');
181:         $u = self::user($pdo, $uid);
182:         $hidden = [];
183:         // each kind the way it's hidden: [table, id column, owner column, column, its hidden value, which rows]
184:         foreach ([
185:             ['journeys', 'id', 'author_id', 'status', 'hidden', "status <> 'hidden'"],
186:             ['journey_comments', 'id', 'user_id', 'status', 'hidden', "status <> 'hidden'"],
187:             ['content_comments', 'id', 'user_id', 'status', 'hidden', "status <> 'hidden'"],
188:             ['place_stories', 'id', 'author_id', 'status', 'draft', "status = 'published'"],
189:             ['posts', 'id', 'user_id', 'is_deleted', '1', 'is_deleted = 0'],
190:             ['experiences', 'id', 'author_id', 'status', 'hidden', "status <> 'hidden'"],
191:             ['member_profiles', 'user_id', 'user_id', 'published', '0', 'published = 1'],
192:             ['community_groups', 'id', 'owner_id', 'status', 'hidden', "status = 'active'"],
193:         ] as [$t, $idc, $own, $col, $to, $where]) {
194:             foreach (self::q($pdo, "SELECT `$idc` AS id, `$col` AS v FROM `$t` WHERE `$own` = ? AND $where", [$uid]) as $r) $hidden[] = [$t, $idc, $r['id'], $col, $r['v'], $to];
195:         }
196:         foreach (self::q($pdo, 'SELECT id FROM community_posts WHERE user_id = ? AND removed_at IS NULL', [$uid]) as $r) $hidden[] = ['community_posts', 'id', $r['id'], 'removed_at', null, 'now'];
197:         $token = bin2hex(random_bytes(20));
198:         $pdo->beginTransaction();
199:         try {
200:             foreach ($hidden as [$t, $idc, $id, $col, , $to]) {
201:                 $pdo->prepare("UPDATE `$t` SET `$col` = " . ($to === 'now' ? 'NOW()' : '?') . " WHERE `$idc` = ?")->execute($to === 'now' ? [$id] : [$to, $id]);
202:             }
203:             $pdo->prepare("INSERT INTO account_closures (user_id, requested_at, delete_after, status, hidden, token_hash) VALUES (?, NOW(), DATE_ADD(NOW(), INTERVAL " . self::GRACE_DAYS . " DAY), 'closing', ?, ?)
204:                            ON DUPLICATE KEY UPDATE requested_at = NOW(), delete_after = VALUES(delete_after), status = 'closing', hidden = VALUES(hidden), token_hash = VALUES(token_hash), restored_at = NULL, deleted_at = NULL")
205:                 ->execute([$uid, json_encode($hidden), hash('sha256', $token)]);
206:             $pdo->prepare('UPDATE users SET session_version = session_version + 1, updated_at = NOW() WHERE id = ?')->execute([$uid]);   // signed out everywhere
207:             $pdo->commit();
208:         } catch (Throwable $e) {
209:             $pdo->rollBack();
210:             error_log('account close ' . $uid . ': ' . $e->getMessage());
211:             throw new RuntimeException('Your account couldn\'t be closed just now. Nothing was changed. Try again, or email support@traversence.com.');
212:         }
213:         $after = (string) (self::closing($pdo, $uid)['delete_after'] ?? '');
214:         return ['email' => (string) $u['email'], 'delete_after' => $after, 'token' => $token];
215:     }

217:     /** Opens a closing account again: what close() hid comes back (where nobody has changed it since). */
218:     public static function restore(PDO $pdo, int $uid): void
219:     {
220:         $c = self::closing($pdo, $uid);
221:         if (!$c) throw new RuntimeException('There\'s nothing to restore: this account isn\'t closing.');
222:         $pdo->beginTransaction();
223:         try {
224:             foreach ((array) json_decode((string) $c['hidden'], true) as $h) {
225:                 [$t, $idc, $id, $col, $was, $to] = $h + [null, null, null, null, null, null];
226:                 if (!preg_match('/^[a-z_]+$/', (string) $t . $idc . $col)) continue;
227:                 if ($to === 'now') $pdo->prepare("UPDATE `$t` SET `$col` = NULL WHERE `$idc` = ? AND `$col` IS NOT NULL")->execute([$id]);
228:                 else $pdo->prepare("UPDATE `$t` SET `$col` = ? WHERE `$idc` = ? AND `$col` = ?")->execute([$was, $id, $to]);
229:             }
230:             $pdo->prepare("UPDATE account_closures SET status = 'restored', restored_at = NOW(), token_hash = NULL WHERE user_id = ?")->execute([$uid]);
231:             $pdo->commit();
232:         } catch (Throwable $e) {
233:             $pdo->rollBack();
234:             error_log('account restore ' . $uid . ': ' . $e->getMessage());
235:             throw new RuntimeException('Your account couldn\'t be restored just now. Try again, or email support@traversence.com.');
236:         }
237:     }

239:     /** Day 30: the full deletion for every closure that's due. Returns how many. */
240:     public static function sweep(PDO $pdo): int
241:     {
242:         if (!self::ready($pdo)) return 0;
243:         $n = 0;
244:         foreach (self::q($pdo, "SELECT user_id FROM account_closures WHERE status = 'closing' AND delete_after <= NOW() LIMIT 20", []) as $r) {
245:             $uid = (int) $r['user_id'];
246:             try {
247:                 $email = self::run($pdo, $uid);
248:                 $pdo->prepare("UPDATE account_closures SET status = 'deleted', deleted_at = NOW(), token_hash = NULL, hidden = NULL WHERE user_id = ?")->execute([$uid]);
249:                 self::mail($email, 'Your Traversence account is deleted',
250:                     "The 30 days are up, and your Traversence account has been deleted, as you asked.\n\nYour profile, address book, connections, saved places and settings are gone, "
251:                     . "and what you wrote (journeys, stories, posts and comments) has been erased.\n\nThank you for being part of Traversence.");
252:                 $n++;
253:             } catch (Throwable $e) { error_log('account sweep ' . $uid . ': ' . $e->getMessage()); }   // tried again on the next sweep
254:         }
255:         return $n;
256:     }

258:     /** sweep(), from page loads, at most every 30 minutes. */
259:     public static function tickSoon(PDO $pdo): void
260:     {
261:         $f = dirname(__DIR__) . '/storage/closures.tick';
262:         $last = @filemtime($f);
263:         if ($last !== false && $last > time() - 1800) return;
264:         @touch($f);
265:         try { self::sweep($pdo); } catch (Throwable $e) { error_log('closures tick: ' . $e->getMessage()); }
